Solution exercice 5.5

On calcule le paramètre d’intérêt et les probabilités d’inclusion \(\pi_{kI}, k\in U_I\) proportionnelles à la taille de la grappe, donnée dans ce cas par le BVQ_N.

require(sampling)
require(survey)
data(rec99)
##simu: plan SI
##paramètre d'intérêt 
tLOGVAC <- sum(rec99$LOGVAC)

nI <- 4
taille_grappe <- table(rec99$BVQ_N)
sum(taille_grappe)
#[1] 554
pik_rec <- inclusionprobabilities(taille_grappe,nI)

On évalue l’efficacité de cette stratégie par des simulations de Monte-Carlo.

set.seed(123456)
est.grap_sy <- matrix(1,10000,1)
for (i in 1:10000)
{
  grap.rec_sy <- sampling::cluster(rec99,clustername=c("BVQ_N"),
                                   size=nI,method="systematic",pik=pik_rec) 
  grap.data_sy <- getdata(rec99,grap.rec_sy)
  
  ech.grap_sy <- svydesign(ids=~BVQ_N, probs = ~Prob, data=grap.data_sy ) 
  est.grap_sy[i] <- svytotal(~LOGVAC,ech.grap_sy)[1]
}
#biais relatif
br.grapsy <- (mean(est.grap_sy)-tLOGVAC)/tLOGVAC
br.grapsy
#[1] -0.001939559

#coefficient de variation
cv.grapsy <- sd(est.grap_sy)/mean(est.grap_sy)
cv.grapsy
#[1] 0.2783709

La section 5.5 du tome 1 traite les stratégies basées sur les plans SIC et STSIC avec l’estimateur de Horvitz-Thompson et leur efficacité est donnée dans le tableau 5.4. On peut vérifier que l’efficacité de la stratégie basée sur le plan SIC proportionnel au nombre de communes de chaque grappe est semblable à celle du plan stratifié et meilleure que celle du plan SIC.

Solution exercice 5.6

On calcule le paramètre d’intérêt, le total de la variable \(y\)=Surfacesbois, et on l’estime avec un plan SIC de taille \(n_I=6\), les grappes étant données par les cantons, variable CT.

require(sampling)
require(survey)
data("swissmunicipalities")

swissmunicipalities$CT <- as.factor(swissmunicipalities$CT)
summary(lm(swissmunicipalities$Surfacesbois~swissmunicipalities$CT,data=swissmunicipalities))
#Adjusted R-squared:  0.162 

##paramètre d'intérêt
tbois <- sum(swissmunicipalities$Surfacesbois)
###############################
###plan srswor des cantons
###dans la pop suisse des villes
################################

NI <- length(levels(swissmunicipalities$CT)); nI <- 6
set.seed(123456)
grap.swiss <- sampling::cluster(swissmunicipalities,
                                clustername="CT",size=nI,method="srswor")
grap.data <- getdata(swissmunicipalities,grap.swiss)

nech <- nrow(grap.data)
ech.grap <- svydesign(id=~1,weights=rep(NI/nI,nech),
                      data=grap.data,fpc=rep(nI/NI,nech))
svytotal(~Surfacesbois,ech.grap) 
#              total    SE
#Surfacesbois 838457 45045

On évalue l’effcacité de cette stratégie via des simulations de Monte-Carlo.

set.seed(123456)
n.sim <- 10000
est.grap_si <- matrix(1,n.sim,1)
for (i in 1:n.sim)
{
  grap.swiss <- sampling::cluster(swissmunicipalities,
                                  clustername="CT",size=nI,method="srswor")
  grap.data <- getdata(swissmunicipalities,grap.swiss)
  
  nech <- nrow(grap.data)
  ech.grap <- svydesign(id=~1,weights=rep(NI/nI,nech),
                        data=grap.data,fpc=rep(nI/NI,nech))
  est.grap_si[i]<-svytotal(~Surfacesbois,ech.grap)[1] 
}
br.grap <- (mean(est.grap_si)-tbois)/tbois
br.grap
#[1] -0.002889166
cv_grap_si <- sd(est.grap_si)/mean(est.grap_si)
cv_grap_si
#[1] 0.389648

Solution exercice 5.7

Suite de l’exercice 5.6

require(sampling)
require(survey)
data("swissmunicipalities")

##paramètre d'intérêt
tbois <- sum(swissmunicipalities$Surfacesbois)

NI <- length(levels(swissmunicipalities$CT)); nI <- c(6,8,12,16)
set.seed(123456)
n.sim <- 10000
est.grap_si <- matrix(1,n.sim,4)
for (j in 1:length(nI))
{
  for (i in 1:n.sim)
    {
    grap.swiss <- sampling::cluster(swissmunicipalities,
                                  clustername="CT",size=nI[j],method="srswor")
    grap.data <- getdata(swissmunicipalities,grap.swiss)
    nech <- nrow(grap.data)
    ech.grap <- svydesign(id=~1,weights=rep(NI/nI[j],nech),
                        data=grap.data,fpc=rep(nI[j]/NI,nech))
    est.grap_si[i,j]<-svytotal(~Surfacesbois,ech.grap)[1] 
  }
}
br <- (apply(est.grap_si,2,mean)-tbois)/tbois
cv <- apply(est.grap_si,2,sd)/apply(est.grap_si,2,mean)

  1. Comparaison de stratégies via de simulations de Monte-Carlo:
N <- nrow(emploi); n <- 100

n.sim <- 10000
taux.si <- taux.be <- taux.sy <- numeric(n.sim)
p.agri.si <- p.agri.be <- p.agri.sy <- numeric(n.sim) 

#plan SI
set.seed(123456)
for (i in 1:n.sim)
{
  s.emploi.SI <- srswor(n,N)
  ech.emploi.si <- svydesign(id=~1, weights=rep(N/n,n),
                             fpc=rep(n/N,n),data=emploi[s.emploi.SI==1,])
  
  A.SI <- svymean(~AGRI,ech.emploi.si)
  p.agri.si[i] <- A.SI[[1]]
  taux.si[i] <- pAGRI>= A.SI[[1]]-1.96*SE(A.SI) && pAGRI<= A.SI[[1]]+1.96*SE(A.SI)
}

Solution exercice 2.18

On commence par calculer les paramètres d’intérêt, le total de la variable \(y\) = LOGVAC et la variance de l’estimateur de Horvitz-Thompson de ce total selon les plans de sondage SI de taille \(n\)=70 et BE d’une taille moyenne \(\mathbb{E}(n_s)=70.\)

N <- nrow(rec99); n <- 70

##paramètres d'intérêt
ty <- sum(rec99$LOGVAC)
ty
##[1] 10768
varHT.SI <- N^2*(1/n-1/N)*var(rec99$LOGVAC)
varHT.SI
##[1] 4230802

varHT.BE <- (N/n-1)*sum(rec99$LOGVAC^2)
varHT.BE
##[1] 5670295
  1. et 2. On compare, via des simulations de Monte-Carlo, les stratégies SI et BE pour estimer le total de \(y\)=LOGVAC d’un point de vue du taux de couverture sous les deux plans de sondage. On calcule, à partir de mêmes 10000 estimations des variances obtenues selon les deux plans de sondage, les biais relatifs ainsi que les coefficients de variations des estimateurs de la variance.
##plan SI
n.sim <- 10000
set.seed(123456)
estvar.si <- numeric(n.sim)
taux.si <- numeric(n.sim)
for (i in 1:n.sim)
{
  si.rec <- srswor(n,N)
  ech.si <- svydesign(id=~CODE_N, weights=rep(N/n,n),
                      fpc=rep(n/N,n),data=rec99[si.rec==1,])
  A <- svytotal(~LOGVAC,ech.si)
  estvar.si[i] <- (SE(A))^2
  taux.si[i] <-  ty>= A[[1]]-1.96*SE(A) && ty<= A[[1]]+1.96*SE(A)
}

##taux de couverture de ty, plan SI
taux.couverture <- mean(taux.si)
taux.couverture
##[1] 0.9014



##biais relatif de l'estimateur de la variance
br.estvar.si <- (mean(estvar.si)-varHT.SI)/varHT.SI
br.estvar.si
#[1] 0.00478774

##coefficient de variation  de l'estimateur de la variance
cv.estvar.si<-sd(estvar.si)/mean(estvar.si)
cv.estvar.si
##[1] 0.6198478

##plan BE de paramètre pi=n/N
pik <- rep(n/N, N)
n.sim <- 10000
set.seed(123456)
estvar.be <- numeric(n.sim)
taux.be <- numeric(n.sim)
for (i in 1:n.sim)
{
  s.be <- UPpoisson(pik)
  rec.be <- rec99[s.be==1,]
  est.be <- (N/n) * sum(rec.be$LOGVAC)
  estvar.be[i] <- (N/n-1)*(N/n)*sum(rec.be$LOGVAC^2)
  taux.be[i] <-  ty>= est.be-1.96*sqrt(estvar.si[i]) && ty<= est.be +1.96*sqrt(estvar.si[i])
}

##taux de couverture de ty
taux.couverture <- mean(taux.be)
taux.couverture
##[1] 0.8599

##coefficient de la variation  de l'estimateur de la variance
cv.estvar.be<-sd(estvar.be)/mean(estvar.be)
cv.estvar.be
##[1] 0.5546522

##biais relatif de l'estimateur de la variance
br.estvar.be <- (mean(estvar.be)-varHT.BE)/varHT.BE
br.estvar.be
##[1] 0.004447049

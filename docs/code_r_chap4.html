# fonction arrondi pour calculer les tailles nh des échantillons sh
# dans le cas d'un plan stratifié

arrondi <- function(a)
{
  o <- order(a-floor(a),decreasing=T) #ordonne selon
  #parties décimales
  n <- ceiling(sum(a)) #calcule taille totale de l’échantillon
  #avec prise en compte des arrondis éventuels au niveau des nh
  nbar = n-sum(floor(a)) #calcule le nombre de nh
  #à arrondir à l’unité par excès
  nouv <- floor(a) #arrondit tous les nh à l’unité par défaut
  if (nbar!=0){nouv[o[1:nbar]] <- nouv[o[1:nbar]] + 1}
  #arrondit certains nh par excès si nécessaire
  #(les nh avec les plus faibles écarts à l’entier supérieur)
  if (any(nouv == 0))
  {print("Attention, une strate est à tirage nul")}
  nouv
}

set.seed(123456)
n.sim <- 10000
p.AGRI.STSIprop <- numeric(n.sim)
taux.STSIprop <- numeric(n.sim)
for (i in 1:n.sim)
  {
  stsi.prop <- sampling::strata(emploi.order,"TURbis",size=n.prop,
                                method="srswor")
  stsi.data.prop <- getdata(emploi.order,stsi.prop)
  ech.stsi.prop <- svydesign(id=~1,strata=~TURbis,
                                 weights=rep(Nhbis/n.prop,n.prop),
                                 fpc=rep(n.prop/Nhbis,n.prop),
                                 data=stsi.data.prop)
  A.stsi.prop <- svymean(~AGRI,ech.stsi.prop)
  p.AGRI.STSIprop[i]<-A.stsi.prop[[1]]
  taux.STSIprop[i] <- pAGRI>= A.stsi.prop[[1]]-1.96*SE(A.stsi.prop) && pAGRI<= A.stsi.prop[[1]]+1.96*SE(A.stsi.prop)
}

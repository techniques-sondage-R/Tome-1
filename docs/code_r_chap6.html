Solution exercice 6.4

require(sampling)
require(survey)
data(rec99)
##données sur la population 
NI <- length(table(rec99$BVQ_N))

##parametres d'interêt
##1. la population totale
tpop <- sum(rec99$POPSDC99)

  1. On calcule la variance de l’estimateur HT, voir la section 6.2.1 du tome 1 pour le plan à deux degrés SI,SI de taille nI=14 UP et ni=5 US dans chaque UP sélectionnée:
nI <- 14; ni <- 5
tUI <- tapply(rec99$POPSDC99, rec99$BVQ_N, sum)

On prépare la table pour le tirage à deux degrés:

o <- cbind.data.frame(table(rec99$BVQ_N), NI)
names(o) <- c("BVQ_N", "Ni", "NI")
rec_2deg <- merge(rec99,o)

rec_2deg.ord <- rec_2deg[order(rec_2deg$BVQ_N,rec_2deg$CODE_N),]

On évalue l’efficacité (via des simulations de Monte-Carlo) des estimateurs de la variance (de type Horvitz-Thomspon et l’estimateur simplifié) de l’estimateur de Horvitz-Thompson de la population totale:

N1 <- rep(NI, nI*ni)
n.sim <- 10000
set.seed(123456)
est.SISI  <- var.SISI<- var.SISI.SIMP <- numeric(n.sim)
for (i in 1:n.sim)
{
  deg.rec <- mstage(rec_2deg.ord,stage=c("cluster","cluster"),
                    varnames=list("BVQ_N","CODE_N"),
                    size=list(nI ,c(rep(ni,nI))), 
                    method=c("srswor","srswor"))
  deg2.rec <- getdata(rec_2deg.ord,deg.rec)[[2]]
  ech.2deg <- svydesign(id=~BVQ_N+CODE_N,fpc=~N1+Ni,data=deg2.rec)
  est.SISI[i] <- svytotal(~POPSDC99, ech.2deg)[1]
  var.SISI[i] <- (SE(svytotal(~POPSDC99, ech.2deg)))^2
  ##est de la variance simplifiée
  ech.2degSIMP <- svydesign(id=~BVQ_N,fpc=~N1,weights=(NI/nI)*(deg2.rec$Ni/ni), 
                            data=deg2.rec)
  
  var.SISI.SIMP[i] <- (SE(svytotal(~POPSDC99, ech.2degSIMP)))^2
}
#biais relatif de l'estimateur de la population
br_est <- (mean(est.SISI)-tpop)/tpop
br_est
#[1] 0.002093423

#biais relatif de l'estimateur de la variance
br_estvar <- (mean(var.SISI)-var_tot2deg)/var_tot2deg
br_estvar
#[1] 0.005610094

#biais relatif de l'estimateur de la variance simplifié
br_estvarSIMP <- (mean(var.SISI.SIMP)-var_tot2deg)/var_tot2deg
br_estvarSIMP
#[1] -0.1243742

On prépare la table pour le tirage à deux degrés:

o <- cbind.data.frame(table(rec99$BVQ_N), NI)
names(o) <- c("BVQ_N", "Ni", "NI")
rec_2deg <- merge(rec99,o)

rec_2deg.ord <- rec_2deg[order(rec_2deg$BVQ_N,rec_2deg$CODE_N),]

N1 <- rep(NI, nI*ni)

On désire évaluer (par des simulations de Monte-Carlo) l’efficacité de deux estimateurs de la variance et calculer les taux de couverture de tLOGVAC.

set.seed(123456)
n.sim <- 10000
var.SISI <- var.SISI.V0 <- numeric(n.sim)
taux.var <- taux.var.V0 <- numeric(n.sim)
for (i in 1:n.sim)
{
  deg.rec <- mstage(rec_2deg.ord,stage=c("cluster","cluster"),
                    varnames=list("BVQ_N","CODE_N"),
                    size=list(nI ,c(rep(ni,nI))), 
                    method=c("srswor","srswor"))
  deg2.rec <- getdata(rec_2deg.ord,deg.rec)[[2]]
  ech.2deg <- svydesign(id=~BVQ_N+CODE_N,fpc=~N1+Ni,data=deg2.rec)
  A <- svytotal(~LOGVAC, ech.2deg)
  ##est de la variance
  var.SISI[i] <- (SE(A))^2
  ##taux de couverture
  taux.var[i] <- tLOGVAC >= A[[1]]-1.96*SE(A) && tLOGVAC<= A[[1]]+1.96*SE(A)
  ##est de la variance simplifiée
  ech.2degSIMP <- svydesign(id=~BVQ_N,fpc=~N1,weights=(NI/nI)*(deg2.rec$Ni/ni), 
                            data=deg2.rec)
  A0 <- svytotal(~LOGVAC, ech.2degSIMP)
  var.SISI.V0[i] <- (SE(A0))^2/(1-nI/NI)
  ##taux de couverture
  taux.var.V0[i] <- tLOGVAC >= A[[1]]-1.96*sqrt(var.SISI.V0[i])  && tLOGVAC<=A[[1]]+1.96*sqrt(var.SISI.V0[i])
}

#biais relatif de l'estimateur de la variance
br_estvar <- (mean(var.SISI)-var_tot2deg)/var_tot2deg
br_estvar
#[1] 0.00812897

#biais relatif de l'estimateur de la variance simplifié
#en mimant un plan avec remise
br_estvar.SISI.V0 <- (mean(var.SISI.V0)-var_tot2deg)/var_tot2deg
br_estvar.SISI.V0
#[1] 0.2677377

4 et 5. avec simulations

n.sim <- 10000
taux.pps <- numeric(n.sim)
est.pps <- numeric(n.sim) 

set.seed(123456)
for (i in 1:n.sim)
{
  R <- UPmultinomial(m*pk)
  ech.pps.o <- rep((1:N)[R!=0],R[R!=0])
  ech.pps <- svydesign(id=~1,weights=recs[ech.pps.o,"poids"], data=recs[ech.pps.o,])
  A <- svytotal(~LOGVAC, ech.pps)
  est.pps[i] <- A[[1]]
  taux.pps[i] <- tLOGVAC >= A[[1]]-1.96*SE(A) && tLOGVAC<= A[[1]]+1.96*SE(A)
}
#biais relatif
br.pps <- (mean(est.pps)-tLOGVAC)/tLOGVAC
br.pps 
#[1] 0.001077128

#coefficient de variation
cv.pps <- sd(est.pps)/mean(est.pps)
cv.pps
#[1] 0.07612369
#taux de couverture 
taux <- mean(taux.pps)
taux
#[1] 0.9395

Solution exercice 3.13

On calcule le paramètre d’intérêt et les probabilités d’inclusion, proportionnelles au nombre de logements vacants d’une commune.

N <- nrow(rec99)
tLOGVAC <- sum(rec99$LOGVAC)
n <- 70
pik <- inclusionprobabilities(rec99$LOG,n)
poids <- 1/pik
pk <- pik/n

On étudie le comportement de l’estimateur simplifié de la variance de l’estimateur de Horvitz-Thompson via des simulations de Monte-Carlo.

n.sim <- 10000
set.seed(123456)
tHT <- varsimp.pips <- numeric(n.sim)

for (i in 1:n.sim)
{
  s <- UPsystematic(pik)
  tHT[i] <- sum(rec99$LOGVAC[s==1]*poids[s==1])
  varsimp.pips[i] <- (sum((rec99$LOGVAC[s==1]/pk[s==1])^2)-n*tHT[i]^2)/(n*(n-1))
}
br.HT <- (mean(tHT)-tLOGVAC)/tLOGVAC
br.HT 
#[1] 0.000456236

cv.varsimp <- sd(varsimp.pips)/mean(varsimp.pips)
cv.varsimp
#[1] 0.2487124

On compare avec la variance empirique de Monte-Carlo des estimations car la vraie variance de l’estimateur de Horvitz-Thompson est difficile à calculer pour un plan proportionnel à la taille sans remise.
